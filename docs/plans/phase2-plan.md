## 3. 主要な型定義（案）

```ts
// ---- sim: ルールと設定 ----
type PartId =
  | /* 既存9種 */ 'merger'
  | 'chainMeter'
  | 'spreader'
  | 'copier'
  | 'reflector'
  | 'turntable'
  | 'oiler'
  | 'coil'
  | 'solar'
  | 'inspector'
  | 'piggyBank';
type Rarity = 'common' | 'uncommon' | 'rare';

interface RuleSet {
  tickLimit: number;
  switchSignalValue: number;
  maxActivations: Record<PartId, number | null>;
  params: PartParams; // ギア倍率・プレス係数・検品台倍率など、パーツ別の数値
  blockedCells: number[]; // 使用不可マス（ボス）
  dockDivisor: number; // 出荷口の加算を割る値（通常1、ボスで2）
  maxIncomePerSim: number; // 貯金箱の収入上限
}

interface RunConfig {
  configVersion: 1;
  board: { width: number; height: number };
  rules: RuleSet; // 基本＋メタ（ボス抜き）
  economy: {
    prices: Record<PartId, number>;
    shopPool: { partId: PartId; weight: number }[]; // 解放済みパーツのみ
    offersPerShift: number;
    reroll: { baseCost: number; costStep: number };
    refundPercent: number;
  };
  shifts: { quota: number; budget: number; clearReward: number; kind: 'normal' | 'boss' }[];
  bossPlan: (BossModifierId | null)[]; // シフトごと。ラン開始時に抽選
}
type BossModifierId = 'lowOil' | 'repairWork' | 'strictInspection' | 'shortShift' | 'partShortage';

interface RunState {
  seed: number;
  config: RunConfig;
  shiftIndex: number; // 0..8（day = floor(i/3), 時間帯 = i % 3）
  phase: 'building' | 'cleared' | 'failed';
  budget: number;
  board: Board;
  inventory: Partial<Record<PartId, number>>;
  shop: ShopOffer[];
  rerollCount: number; // このシフトのリロール回数
  trialCount: number; // このシフトの試運転回数（試運転シード用）
  history: ShiftRecord[];
}

// ---- sim: 出力の追加 ----
interface SimResult {
  score: Score;
  income: number;
  events: SimEvent[];
  stats: SimStats;
}
type SimEvent =
  /* 既存 */
  | { tick; type: 'absorb'; signalId; x; y } // 合流炉が信号を溜めた
  | { tick; type: 'income'; x; y; amount; total } // 貯金箱
  | { tick; type: 'halt'; reason: 'tickLimit' }; // 途中打ち切り（途切れた理由の表示用）
// vanish の reason に 'blocked' を追加

  version: 2;
  run: RunState | null;
  meta: MetaProgress;

interface MetaProgress {
  unlockedParts: PartId[];
  boardLevel: 0 | 1 | 2; // 7×7 / 8×8 / 9×9
  records: {
    totalShipped: string;
    bestShiftReached: number;
    bestChain: number;
    runsPlayed: number;
    clears: number;
  };
}
// v1→v2 変換: 進行中ランを現行 balance の RunConfig で包み、shiftIndex 0..2 → 1日目 朝・昼・夜 に対応させる。meta は初期値
```

## 4. 追加パーツ11種

| ID         | 名前           | 軸     | レア度     | 発動回数                | 挙動                                                               | 狙っているシナジー                                         |
| ---------- | -------------- | ------ | ---------- | ----------------------- | ------------------------------------------------------------------ | ---------------------------------------------------------- |
| merger     | 合流炉         | 倍率   | アンコモン | 2（1 tick の合流で1回） | 同じ tick に入った信号をすべて合算し、tick の終わりに1本にして送る | 分岐器・散布機で増やした信号をタイミングを揃えて再集約する |
| chainMeter | 連鎖メーター   | 倍率   | レア       | 1                       | 値 ×（その時点の連鎖数÷5＋1）                                      | 長い連鎖の終盤に置くほど強い                               |
| spreader   | 散布機         | 分岐   | コモン     | 1                       | 前・左・右の3方向へ同じ値で送る                                    | 出荷口3つ、または合流炉へ                                  |
| copier     | コピー機       | 分岐   | アンコモン | 1                       | 自身の向きへ同じ値を2連射（2発目は1 tick 遅れ）                    | コンベアなど複数回発動するパーツ、タイミングずらし         |
| reflector  | 反射板         | 再発動 | コモン     | 2                       | 来た方向へ跳ね返す                                                 | 往復でギア・コイルを2回通す（相手にも発動回数が必要）      |
| turntable  | 回転台         | 再発動 | アンコモン | 4                       | 自身の向きへ送り、発動のたびに時計回りに90度回る                   | 反射板と組んで4方向へ順番に配る                            |
| oiler      | 潤滑油タンク   | 再発動 | レア       | ―（反応しない）         | 隣接4マスのパーツの発動回数上限＋1（常時）                         | ギア・プレスの2回発動、ドラム缶の2回爆発                   |
| coil       | 共鳴コイル     | 配置   | コモン     | 1                       | 値 ×（1＋隣接する共鳴コイル数）                                    | コイルを固めて並べる。プレスとは隣接枠を取り合う           |
| solar      | ソーラーパネル | 配置   | コモン     | 1                       | 値 ＋ 周囲8マスの空きマス数                                        | 序盤の小さな値を底上げ。密集させたいプレスと逆の性質       |
| inspector  | 検品台         | 配置   | アンコモン | 1                       | 隣接に出荷口があれば ×3                                            | 出荷口の手前で最後の一押し                                 |
| piggyBank  | 貯金箱         | 経済   | アンコモン | 3                       | そのまま送り、次シフトの予算＋1（1回のシミュレーションで最大＋3）  | 序盤に入れて後半の購入力に変える                           |

既存パーツのレア度: コンベア・出荷口・ポンコツロボ・ギア・分岐器はコモン、プレス機・ドラム缶はアンコモン、再起動装置はレア。
ショップの出現重みは レア度ごとの既定値（コモン10／アンコモン5／レア2）とし、パーツ別に上書きできるようにする。

**メタ進行の初期解放（2b）**: 既存9種＋コモン4種（散布機・反射板・共鳴コイル・ソーラーパネル）＝13種。残り7種と盤面拡張は実績で解放する（例: 連鎖15以上で合流炉、2日目到達で回転台、全9シフトクリアで 8×8）。
